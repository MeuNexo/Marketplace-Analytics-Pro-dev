## Tarefa 2 — Criar `src/hooks/useMLAdsDerivedMetrics.ts` (arquivo NOVO)

Hook que enriquece `AdsProductStat[]` com métricas calculadas via queries adicionais.

### Interface de retorno

```typescript
export interface EnrichedAdsProduct extends AdsProductStat {
  acos: number | null;
  cvr: number | null;
  tacos: number | null;
  organic_revenue: number | null;
  share_ads_pct: number | null;
  acos_breakeven: number | null;
  sov: null; // não disponível
}

export interface AdsGlobalDerived {
  total_revenue: number;
  organic_revenue: number;
  tacos_global: number | null;
}
```

### Implementação

```typescript
import { useQuery } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";
import { useMLStore } from "@/contexts/MLStoreContext";
import { useOrganization } from "@/contexts/OrganizationContext";
import type { AdsProductStat } from "./useMLAds";

export function useMLAdsDerivedMetrics(
  products: AdsProductStat[],
  totalSpend: number,
  totalAttributedRevenue: number,
  dateFrom: string,
  dateTo: string,
) {
  const { resolvedMLUserIds } = useMLStore();
  const { currentOrg } = useOrganization();

  // 1. Buscar receita total do período (ml_daily_cache)
  const { data: dailyCache } = useQuery({
    queryKey: ["ml_ads_daily_revenue", currentOrg?.id, resolvedMLUserIds, dateFrom, dateTo],
    queryFn: async () => {
      if (!currentOrg?.id) return { total_revenue: 0, total_orders: 0 };
      const { data } = await supabase
        .from("ml_daily_cache")
        .select("approved, qty")
        .eq("organization_id", currentOrg.id)
        .in("ml_user_id", resolvedMLUserIds)
        .gte("date", dateFrom)
        .lte("date", dateTo);
      const total_revenue = (data ?? []).reduce((s, r) => s + (r.approved ?? 0), 0);
      const total_orders  = (data ?? []).reduce((s, r) => s + (r.qty ?? 0), 0);
      return { total_revenue, total_orders };
    },
    enabled: !!currentOrg?.id && resolvedMLUserIds.length > 0,
    staleTime: 2 * 60 * 1000,
  });

  // 2. Buscar custos unitários para acos_breakeven (product_costs)
  const itemIds = products.map((p) => p.item_id).filter(Boolean);
  const { data: costsData } = useQuery({
    queryKey: ["ml_ads_product_costs", currentOrg?.id, itemIds],
    queryFn: async () => {
      if (!currentOrg?.id || !itemIds.length) return new Map<string, number>();
      const { data } = await supabase
        .from("product_costs")
        .select("item_id, unit_cost")
        .eq("organization_id", currentOrg.id)
        .in("item_id", itemIds)
        .not("unit_cost", "is", null);
      const map = new Map<string, number>();
      for (const r of data ?? []) map.set(r.item_id, r.unit_cost);
      return map;
    },
    enabled: !!currentOrg?.id && itemIds.length > 0,
    staleTime: 5 * 60 * 1000,
  });

  // 3. Enriquecer produtos
  const total_revenue = dailyCache?.total_revenue ?? 0;
  const total_orders  = dailyCache?.total_orders ?? 0;

  const enriched: EnrichedAdsProduct[] = products.map((p) => {
    const acos = p.spend > 0 && p.attributed_revenue > 0
      ? Math.round((p.spend / p.attributed_revenue) * 10000) / 100
      : null;
    const cvr = p.clicks > 0
      ? Math.round((p.attributed_orders / p.clicks) * 10000) / 100
      : null;
    const organic_revenue = p.attributed_revenue != null
      ? null // sem dado por produto — tacos calculado no global
      : null;
    const tacos = null; // TACoS por produto requer receita total por produto — não disponível
    const share_ads_pct = total_orders > 0
      ? Math.round((p.attributed_orders / total_orders) * 10000) / 100
      : null;
    const unit_cost = costsData?.get(p.item_id);
    const avg_price = p.attributed_orders > 0 ? p.attributed_revenue / p.attributed_orders : 0;
    const acos_breakeven = unit_cost != null && avg_price > 0
      ? Math.round(((avg_price - unit_cost) / avg_price) * 10000) / 100
      : null;

    return { ...p, acos, cvr, tacos, organic_revenue, share_ads_pct, acos_breakeven, sov: null };
  });

  // 4. Global derived
  const organic_revenue = total_revenue > 0 ? total_revenue - totalAttributedRevenue : 0;
  const tacos_global = total_revenue > 0 && totalSpend > 0
    ? Math.round((totalSpend / total_revenue) * 10000) / 100
    : null;

  const global: AdsGlobalDerived = { total_revenue, organic_revenue, tacos_global };

  return { enriched, global };
}
```

**Nota:** TACoS por produto individual não é calculável sem receita orgânica por produto. A tabela de produtos mostrará TACoS como "—". TACoS Global (gasto / receita_total_loja) é calculado no `global`.
