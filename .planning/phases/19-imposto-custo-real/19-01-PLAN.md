case "lucro_real": {
  const intra = Number(config.lr_icms_aliquota_intra ?? config.lr_icms_debito ?? 0);
  const interSulSE = Number(config.lr_icms_aliquota_inter_sul_sudeste ?? 12);
  const interNNECO = Number(config.lr_icms_aliquota_inter_norte_nordeste ?? 7);

  const orig = config.uf_origem?.toUpperCase() ?? null;
  const dest = ufDestino?.toUpperCase() ?? null;

  // Determinar alíquota ICMS pela UF destino
  let icmsAliq = intra; // padrão: intraestadual
  if (orig && dest && orig !== dest) {
    icmsAliq = isReducedInterstateDest(dest) ? interNNECO : interSulSE;
  }

  // Fórmula Wesley:
  // Débito ICMS     = Receita × icmsAliq%
  // Base PIS/COFINS = Receita × (1 - icmsAliq%)
  // Débito PIS      = Base × 1,65%
  // Débito COFINS   = Base × 7,60%
  // Total           = ICMS + PIS + COFINS
  const icmsPct = icmsAliq / 100;
  const baseFactor = 1 - icmsPct;           // ex: 0.88 para ICMS 12%
  const pisRate   = 1.65;
  const cofinsRate = 7.60;

  const effectiveRate = icmsAliq + baseFactor * (pisRate + cofinsRate);
  return Math.max(0, effectiveRate);
}

-- Phase 19: Imposto Real + Custo Histórico + Cron Custos
-- 1. Função upsert_orders_preserve_cost — preserva custo_unit existente no ON CONFLICT
-- 2. Recálculo de tax_amount/tax_rate para orders existentes
-- 3. pg_cron sync-tiny-costs-daily

-- ── 1. Função DB para upsert preservando custo_unit ─────────────────────────
-- Chamada por sync-ml-orders v10 em vez de .upsert() direto.
-- ON CONFLICT: preserva custo_unit se já tem valor; só preenche quando NULL.

CREATE OR REPLACE FUNCTION public.upsert_order_preserve_cost(
  p_ml_order_id     TEXT,
  p_ml_user_id      TEXT,
  p_item_id         TEXT,
  p_variation_id    TEXT,
  p_seller_id       TEXT,
  p_user_id         TEXT,
  p_organization_id UUID,
  p_sku             TEXT,
  p_titulo          TEXT,
  p_listing_type    TEXT,
  p_quantidade      INTEGER,
  p_preco_unit      NUMERIC,
  p_comissao        NUMERIC,
  p_frete           NUMERIC,
  p_status          TEXT,
  p_data_pedido     TEXT,
  p_data_pagamento  TEXT,
  p_estado          TEXT,
  p_cidade          TEXT,
  p_comprador       TEXT,
  p_synced_at       TIMESTAMPTZ,
  p_custo_unit      NUMERIC,
  p_tax_rate        NUMERIC,
  p_tax_amount      NUMERIC,
  p_uf_origem       TEXT,
  p_receita_bruta   NUMERIC,
  p_receita_liquida NUMERIC,
  p_marca           TEXT
) RETURNS VOID LANGUAGE plpgsql AS $$
BEGIN
  INSERT INTO public.orders (
    ml_order_id, ml_user_id, item_id, variation_id,
    seller_id, user_id, organization_id, sku, titulo, listing_type,
    quantidade, preco_unit, comissao, frete, status,
    data_pedido, data_pagamento, estado, cidade, comprador,
    synced_at, custo_unit, tax_rate, tax_amount, uf_origem,
    receita_bruta, receita_liquida, marca
  ) VALUES (
    p_ml_order_id, p_ml_user_id, p_item_id, p_variation_id,
    p_seller_id::uuid, p_user_id::uuid, p_organization_id, p_sku, p_titulo, p_listing_type,
    p_quantidade, p_preco_unit, p_comissao, p_frete, p_status,
    p_data_pedido, p_data_pagamento, p_estado, p_cidade, p_comprador,
    p_synced_at, p_custo_unit, p_tax_rate, p_tax_amount, p_uf_origem,
    p_receita_bruta, p_receita_liquida, p_marca
  )
  ON CONFLICT (ml_order_id, ml_user_id, item_id, variation_id)
  DO UPDATE SET
    seller_id       = EXCLUDED.seller_id,
    titulo          = EXCLUDED.titulo,
    listing_type    = EXCLUDED.listing_type,
    quantidade      = EXCLUDED.quantidade,
    preco_unit      = EXCLUDED.preco_unit,
    comissao        = EXCLUDED.comissao,
    frete           = EXCLUDED.frete,
    status          = EXCLUDED.status,
    data_pedido     = EXCLUDED.data_pedido,
    data_pagamento  = EXCLUDED.data_pagamento,
    estado          = EXCLUDED.estado,
    cidade          = EXCLUDED.cidade,
    comprador       = EXCLUDED.comprador,
    synced_at       = EXCLUDED.synced_at,
    -- custo_unit: preserva se já tem valor; preenche se null
    custo_unit      = CASE
                        WHEN orders.custo_unit IS NOT NULL THEN orders.custo_unit
                        ELSE EXCLUDED.custo_unit
                      END,
    tax_rate        = EXCLUDED.tax_rate,
    tax_amount      = EXCLUDED.tax_amount,
    uf_origem       = EXCLUDED.uf_origem,
    receita_bruta   = EXCLUDED.receita_bruta,
    receita_liquida = EXCLUDED.receita_liquida,
    marca           = EXCLUDED.marca;
END;
$$;

-- Permissão para service_role chamar a função
GRANT EXECUTE ON FUNCTION public.upsert_order_preserve_cost TO service_role;


-- ── 2. Recálculo tax_amount/tax_rate para orders existentes ─────────────────
-- Usa a nova fórmula Wesley: ICMS + PIS×(1-ICMS%) + COFINS×(1-ICMS%)
-- Determina alíquota ICMS por UF destino (estado) usando ml_tax_config

DO $$
DECLARE
  cfg RECORD;
BEGIN
  FOR cfg IN SELECT * FROM public.ml_tax_config WHERE regime = 'lucro_real' LOOP
    UPDATE public.orders o
    SET
      tax_rate = CASE
        -- Sem UF origem configurada: usa alíquota intraestadual ou 0
        WHEN cfg.uf_origem IS NULL THEN
          CASE WHEN cfg.lr_icms_aliquota_intra IS NOT NULL
            THEN cfg.lr_icms_aliquota_intra + (1 - cfg.lr_icms_aliquota_intra / 100) * 9.25
            ELSE 0
          END
        -- Com UF origem: determina se inter ou intraestadual
        WHEN o.estado IS NULL OR UPPER(o.estado) = UPPER(cfg.uf_origem) THEN
          -- Intraestadual
          COALESCE(cfg.lr_icms_aliquota_intra, 0) +
          (1 - COALESCE(cfg.lr_icms_aliquota_intra, 0) / 100) * 9.25
        WHEN UPPER(o.estado) IN ('ES','AC','AP','AM','PA','RO','RR','TO',
                                  'AL','BA','CE','MA','PB','PE','PI','RN','SE',
                                  'DF','GO','MT','MS') THEN
          -- Destino Norte/Nordeste/CO/ES → alíquota inter reduzida
          COALESCE(cfg.lr_icms_aliquota_inter_norte_nordeste, 7) +
          (1 - COALESCE(cfg.lr_icms_aliquota_inter_norte_nordeste, 7) / 100) * 9.25
        ELSE
          -- Destino Sul/Sudeste → alíquota inter padrão
          COALESCE(cfg.lr_icms_aliquota_inter_sul_sudeste, 12) +
          (1 - COALESCE(cfg.lr_icms_aliquota_inter_sul_sudeste, 12) / 100) * 9.25
      END,
      tax_amount = CASE
        WHEN o.receita_bruta IS NULL THEN NULL
        ELSE o.receita_bruta * (
          CASE
            WHEN cfg.uf_origem IS NULL THEN
              CASE WHEN cfg.lr_icms_aliquota_intra IS NOT NULL
                THEN cfg.lr_icms_aliquota_intra + (1 - cfg.lr_icms_aliquota_intra / 100) * 9.25
                ELSE 0
              END
            WHEN o.estado IS NULL OR UPPER(o.estado) = UPPER(cfg.uf_origem) THEN
              COALESCE(cfg.lr_icms_aliquota_intra, 0) +
              (1 - COALESCE(cfg.lr_icms_aliquota_intra, 0) / 100) * 9.25
            WHEN UPPER(o.estado) IN ('ES','AC','AP','AM','PA','RO','RR','TO',
                                      'AL','BA','CE','MA','PB','PE','PI','RN','SE',
                                      'DF','GO','MT','MS') THEN
              COALESCE(cfg.lr_icms_aliquota_inter_norte_nordeste, 7) +
              (1 - COALESCE(cfg.lr_icms_aliquota_inter_norte_nordeste, 7) / 100) * 9.25
            ELSE
              COALESCE(cfg.lr_icms_aliquota_inter_sul_sudeste, 12) +
              (1 - COALESCE(cfg.lr_icms_aliquota_inter_sul_sudeste, 12) / 100) * 9.25
          END
        ) / 100
      END
    WHERE o.organization_id = cfg.organization_id
      AND o.tax_rate IS DISTINCT FROM (
        CASE
          WHEN cfg.uf_origem IS NULL THEN 0
          ELSE COALESCE(cfg.lr_icms_aliquota_inter_sul_sudeste, 12) +
               (1 - COALESCE(cfg.lr_icms_aliquota_inter_sul_sudeste, 12) / 100) * 9.25
        END
      );

    RAISE NOTICE 'Recalculated tax for org %: % orders updated', cfg.organization_id,
      (SELECT COUNT(*) FROM public.orders WHERE organization_id = cfg.organization_id);
  END LOOP;
END $$;
