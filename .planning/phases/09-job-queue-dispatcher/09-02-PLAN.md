    1. Verificar estrutura de sync_jobs:
       SELECT column_name, data_type, column_default, is_nullable
       FROM information_schema.columns
       WHERE table_schema = 'public' AND table_name = 'sync_jobs'
       ORDER BY ordinal_position;
       ESPERADO: 12 colunas — id (uuid), organization_id (uuid), ml_user_id (text), job_type (USER-DEFINED), date_from (date), date_to (date), status (USER-DEFINED, default 'pending'), retries (integer, default 0), error_msg (text), started_at (timestamptz), finished_at (timestamptz), created_at (timestamptz)

    2. Verificar ENUMs criados:
       SELECT typname, enumlabel
       FROM pg_type t JOIN pg_enum e ON t.oid = e.enumtypid
       WHERE typname IN ('sync_job_type', 'sync_job_status')
       ORDER BY typname, enumsortorder;
       ESPERADO: sync_job_status com (failed, pending, running, completed); sync_job_type com (daily_cache, inventory, orders)

    3. Verificar indexes:
       SELECT indexname, indexdef
       FROM pg_indexes
       WHERE tablename = 'sync_jobs' AND schemaname = 'public';
       ESPERADO: sync_jobs_pending_created_idx (WHERE status = 'pending') e sync_jobs_open_lookup_idx (WHERE status IN ('pending', 'running'))

    4. Verificar RLS habilitado:
       SELECT tablename, rowsecurity
       FROM pg_tables
       WHERE schemaname = 'public' AND tablename = 'sync_jobs';
       ESPERADO: rowsecurity = true

       SELECT jobname, schedule, active

       WHERE jobname IN ('sync-dispatch-every-30min', 'sync-process-job-every-5min', 'sync-job-retry-watchdog');
       ESPERADO: 3 linhas, todas com active = true; schedules '*/30 * * * *', '*/5 * * * *', '*/5 * * * *'

    6. Verificar dispatch_sync_jobs() e claim_next_sync_job() existem:
       SELECT proname, prosrc IS NOT NULL AS has_body
       FROM pg_proc
       WHERE proname IN ('dispatch_sync_jobs', 'claim_next_sync_job') AND pronamespace = 'public'::regnamespace;
