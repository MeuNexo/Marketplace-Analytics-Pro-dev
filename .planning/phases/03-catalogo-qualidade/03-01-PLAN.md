        expect(calculateEffectiveRate({ regime: "simples_nacional" })).toBe(0)

    describe("Lucro Presumido"):
      it("sums PIS + COFINS + IRPJ + CSLL"):
        expect(calculateEffectiveRate({ regime: "lucro_presumido", lp_pis: 0.65, lp_cofins: 3.0, lp_irpj: 1.2, lp_csll: 1.08 })).toBeCloseTo(5.93)
      it("treats null components as 0"):
        expect(calculateEffectiveRate({ regime: "lucro_presumido", lp_pis: 0.65, lp_cofins: 3.0, lp_irpj: null, lp_csll: null })).toBeCloseTo(3.65)

        expect(calculateEffectiveRate({ regime: "lucro_presumido" })).toBe(0)

    describe("Lucro Real"):
      it("computes debits minus credits"):
        expect(calculateEffectiveRate({ regime: "lucro_real", lr_pis_debito: 1.65, lr_cofins_debito: 7.6, lr_icms_debito: 0, lr_pis_credito: 0.5, lr_cofins_credito: 1.0, lr_icms_credito: 0 })).toBeCloseTo(7.75)

        expect(calculateEffectiveRate({ regime: "lucro_real", lr_pis_debito: 1.0, lr_cofins_debito: 2.0, lr_pis_credito: 2.0, lr_cofins_credito: 3.0 })).toBeCloseTo(-2.0)

        expect(calculateEffectiveRate({ regime: "lucro_real", lr_pis_debito: 1.0, lr_cofins_debito: 1.0, lr_pis_credito: 1.0, lr_cofins_credito: 1.0 })).toBe(0)

        expect(calculateEffectiveRate({ regime: "lucro_real", lr_pis_debito: 1.65, lr_cofins_debito: 7.6, lr_pis_credito: 0, lr_cofins_credito: 0, lr_icms_debito: null, lr_icms_credito: null })).toBeCloseTo(9.25)
      it("returns 0 when all fields are null"):
        expect(calculateEffectiveRate({ regime: "lucro_real" })).toBe(0)

    describe("clampEffectiveRate"):
      it("returns positive rate unchanged"):
        expect(clampEffectiveRate(7.5)).toBe(7.5)
      it("clamps negative rate to 0"):
        expect(clampEffectiveRate(-2.0)).toBe(0)
      it("returns 0 for zero"):
        expect(clampEffectiveRate(0)).toBe(0)
